use monjolo::dynamic_model::model::DynamicModel;
use monjolo::snapshot::Snapshot;

use crate::physics::constants::TepConstants;
use crate::physics::thermo::temperature_from_enthalpy;

const COMPRESSOR_VESSEL_VOLUME: f64 = 5000.0; /* volume do vaso do compressor/condensador [m³] */
const GAS_CONSTANT: f64 = 998.9; /* R em [mmHg·m³/(kmol·K)] */

pub fn new(initial: &Snapshot) -> DynamicModel {
    
    let dm = DynamicModel::new("Compressor");

    let vapor_a = dm.offer_seeded("compressor.state.0", initial.get("state.compressor_vapor.A").unwrap_or(0.0));
    let vapor_b = dm.offer_seeded("compressor.state.1", initial.get("state.compressor_vapor.B").unwrap_or(0.0));
    let vapor_c = dm.offer_seeded("compressor.state.2", initial.get("state.compressor_vapor.C").unwrap_or(0.0));
    let vapor_d = dm.offer_seeded("compressor.state.3", initial.get("state.compressor_vapor.D").unwrap_or(0.0));
    let vapor_e = dm.offer_seeded("compressor.state.4", initial.get("state.compressor_vapor.E").unwrap_or(0.0));
    let vapor_f = dm.offer_seeded("compressor.state.5", initial.get("state.compressor_vapor.F").unwrap_or(0.0));
    let vapor_g = dm.offer_seeded("compressor.state.6", initial.get("state.compressor_vapor.G").unwrap_or(0.0));
    let vapor_h = dm.offer_seeded("compressor.state.7", initial.get("state.compressor_vapor.H").unwrap_or(0.0));
    let energy = dm.offer_seeded("compressor.state.8", initial.get("state.compressor.energy").unwrap_or(0.0));

    let temperature = dm.offer("compressor.temperature");
    let pressure = dm.offer("compressor.pressure");

    let composition_a = dm.offer("compressor.vapor_composition.0");
    let composition_b = dm.offer("compressor.vapor_composition.1");
    let composition_c = dm.offer("compressor.vapor_composition.2");
    let composition_d = dm.offer("compressor.vapor_composition.3");
    let composition_e = dm.offer("compressor.vapor_composition.4");
    let composition_f = dm.offer("compressor.vapor_composition.5");
    let composition_g = dm.offer("compressor.vapor_composition.6");
    let composition_h = dm.offer("compressor.vapor_composition.7");

    let separator_temperature = dm.need("separator.temperature");

    let constants = TepConstants::new();

    dm.on_evaluate(move || {

        let separator_temperature = separator_temperature.get();

        let vapor_a_moles = vapor_a.get();
        let vapor_b_moles = vapor_b.get();
        let vapor_c_moles = vapor_c.get();
        let vapor_d_moles = vapor_d.get();
        let vapor_e_moles = vapor_e.get();
        let vapor_f_moles = vapor_f.get();
        let vapor_g_moles = vapor_g.get();
        let vapor_h_moles = vapor_h.get();
        let total_enthalpy = energy.get();

        let total_vapor_moles = vapor_a_moles
            + vapor_b_moles
            + vapor_c_moles
            + vapor_d_moles
            + vapor_e_moles
            + vapor_f_moles
            + vapor_g_moles
            + vapor_h_moles;

        let fraction_a = vapor_a_moles / total_vapor_moles;
        let fraction_b = vapor_b_moles / total_vapor_moles;
        let fraction_c = vapor_c_moles / total_vapor_moles;
        let fraction_d = vapor_d_moles / total_vapor_moles;
        let fraction_e = vapor_e_moles / total_vapor_moles;
        let fraction_f = vapor_f_moles / total_vapor_moles;
        let fraction_g = vapor_g_moles / total_vapor_moles;
        let fraction_h = vapor_h_moles / total_vapor_moles;

        /* temperature_from_enthalpy pede as 8 frações como &[f64; 8] — assinatura já existente,
        não escolhida aqui; é o único array da função, montado a partir das 8 variáveis nomeadas
        acima, não de um laço.
        */
        let composition = [
            fraction_a, fraction_b, fraction_c, fraction_d, fraction_e, fraction_f, fraction_g, fraction_h,
        ];

        let specific_enthalpy = total_enthalpy / total_vapor_moles;
        let temp = temperature_from_enthalpy(
            &composition,
            separator_temperature,
            specific_enthalpy,
            2,
            &constants,
        );
        let temp_k = temp + 273.15;
        let press = total_vapor_moles * GAS_CONSTANT * temp_k / COMPRESSOR_VESSEL_VOLUME;

        temperature.set(temp);
        pressure.set(press);
        composition_a.set(fraction_a);
        composition_b.set(fraction_b);
        composition_c.set(fraction_c);
        composition_d.set(fraction_d);
        composition_e.set(fraction_e);
        composition_f.set(fraction_f);
        composition_g.set(fraction_g);
        composition_h.set(fraction_h);

        /* Decisão de modelagem inalterada: a derivada real do estado próprio não é calculada
        aqui — quem calcula é Flows, separado, depois deste na sequência.
        */
    });

    dm
}
